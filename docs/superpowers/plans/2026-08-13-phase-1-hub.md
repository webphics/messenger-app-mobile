```ts
export function firstName(displayName: string): string {
  return displayName.trim().split(/\s+/)[0] ?? displayName;
}

export function initials(displayName: string): string {
  const parts = displayName.trim().split(/\s+/).filter(Boolean);
  if (parts.length === 0) return "?";
  if (parts.length === 1) return (parts[0]?.[0] ?? "?").toUpperCase();
  return `${parts[0]?.[0] ?? ""}${parts[1]?.[0] ?? ""}`.toUpperCase();
}

export function formatRelativeTime(iso: string, nowMs = Date.now()): string {
  const then = Date.parse(iso);
  if (Number.isNaN(then)) return "";
  const diffMs = Math.max(0, nowMs - then);
  const minutes = Math.floor(diffMs / 60_000);
  if (minutes < 1) return "just now";
  if (minutes < 60) return `${minutes}m ago`;
  const hours = Math.floor(minutes / 60);
  if (hours < 24) return `${hours}h ago`;
  const days = Math.floor(hours / 24);
  return `${days}d ago`;
}
```

```tsx
import { router } from "expo-router";
import { useState } from "react";
import { Pressable, ScrollView, useWindowDimensions, View } from "react-native";

import { OfferCard } from "@/components/hub/offer-card";
import { Text } from "@/components/ui/text";
import { cn } from "@/lib/cn";
import { chatHref } from "@/lib/chat-route-id";
import { OFFERS } from "@/lib/mock-data";

export function OffersCarousel() {
  const { width } = useWindowDimensions();
  const [index, setIndex] = useState(0);
  const cardWidth = Math.min(width - 48, 360);
  const gap = 12;
  const snap = cardWidth + gap;

  function openOffers() {
    router.push(chatHref("offers"));
  }

  return (
    <View className="mt-6">
      <View className="flex-row items-center justify-between px-1">
        <View className="flex-row items-center gap-2">
          <Text className="font-sans text-base font-semibold text-ink">
            Offers you may like
          </Text>
          <Text className="font-sans text-xs text-muted">Sponsored</Text>
        </View>
        <Pressable
          accessibilityRole="button"
          accessibilityLabel="See all offers"
          onPress={openOffers}
          className="min-h-11 justify-center"
        >
          <Text className="font-sans text-sm font-medium text-primary">
            See all
          </Text>
        </Pressable>
      </View>

      <ScrollView
        horizontal
        pagingEnabled={false}
        snapToInterval={snap}
        decelerationRate="fast"
        showsHorizontalScrollIndicator={false}
        contentContainerClassName="mt-3 gap-3 pr-4"
        onMomentumScrollEnd={(e) => {
          const next = Math.round(e.nativeEvent.contentOffset.x / snap);
          setIndex(Math.min(Math.max(next, 0), OFFERS.length - 1));
        }}
      >
        {OFFERS.map((offer) => (
          <OfferCard key={offer.id} offer={offer} onPress={openOffers} />
        ))}
      </ScrollView>

      <View className="mt-3 flex-row justify-center gap-1.5">
        {OFFERS.map((offer, i) => (
          <View
            key={offer.id}
            className={cn(
              "h-1.5 rounded-full",
              i === index ? "w-4 bg-primary" : "w-1.5 bg-line",
            )}
          />
        ))}
      </View>

      <Pressable
        accessibilityRole="button"
        accessibilityLabel="Join Offers chat"
        onPress={openOffers}
        className="mt-4 min-h-11 items-center justify-center rounded-full bg-primary px-4"
      >
        <Text className="font-sans font-semibold text-white">
          Join Offers chat
        </Text>
      </Pressable>
    </View>
  );
}
```

```tsx
import { CONVERSATIONS, getDevUser, PEOPLE } from "@/lib/mock-data";
import { colors } from "@/lib/tokens";

function assignedAgents() {
  return PEOPLE.filter((p) => p.role === "agent");
}

function latestOwnThread(userId: string) {
  return CONVERSATIONS.filter(
    (c) => c.id !== "offers" && c.participantIds.includes(userId),
  ).sort((a, b) => Date.parse(b.updatedAt) - Date.parse(a.updatedAt))[0];
}

export function ContactHub() {
  const insets = useSafeAreaInsets();
  const me = getDevUser();
  const agents = assignedAgents();
  const recent = latestOwnThread(me.id);
  const otherId = recent?.participantIds.find((id) => id !== me.id);
  const other = PEOPLE.find((p) => p.id === otherId);

  return (
    <ScrollView
      className="flex-1 bg-bg"
      contentContainerClassName="px-4 pb-8"
      contentContainerStyle={{ paddingTop: insets.top + 8 }}
    >
      <View className="flex-row items-center justify-between">
        <Text className="font-sans text-2xl font-bold text-ink">Support</Text>
        <View className="flex-row items-center gap-3">
          <View className="flex-row">
            {agents.map((agent, i) => (
              <Avatar
                key={agent.id}
                className="size-9 border-2 border-bg"
                style={{ marginLeft: i === 0 ? 0 : -10 }}
              >
                <AvatarFallback className="bg-primary-soft">
                  <Text className="font-sans text-xs font-semibold text-primary">
                    {initials(agent.displayName)}
                  </Text>
                </AvatarFallback>
              </Avatar>
            ))}
          </View>
          <Pressable
            accessibilityRole="button"
            accessibilityLabel="Settings"
            onPress={() => router.push("/(contact)/settings")}
            className="size-11 items-center justify-center"
          >
            <Settings color={colors.ink} size={22} />
          </Pressable>
        </View>
      </View>

      <Text className="mt-6 font-sans text-base text-ink-soft">
        Hi {firstName(me.displayName)}
      </Text>
      <Text className="mt-1 font-sans text-2xl font-bold text-ink">
        How can we help?
      </Text>

      <View className="mt-5 flex-row gap-3">
        <Pressable
          accessibilityRole="button"
          accessibilityLabel="Messages"
          onPress={() => router.push("/(contact)/messages")}
          className="min-h-24 flex-1 rounded-2xl bg-surface p-4 shadow-sm shadow-black/5"
        >
          <MessageCircle color={colors.primary} size={22} />
          <Text className="mt-3 font-sans text-base font-semibold text-ink">
            Messages
          </Text>
        </Pressable>
        <Pressable
          accessibilityRole="button"
          accessibilityLabel="Help"
          onPress={() => router.push("/(contact)/help")}
          className="min-h-24 flex-1 rounded-2xl bg-surface p-4 shadow-sm shadow-black/5"
        >
          <HelpCircle color={colors.primary} size={22} />
          <Text className="mt-3 font-sans text-base font-semibold text-ink">
            Help
          </Text>
        </Pressable>
      </View>

      {recent ? (
        <Pressable
          accessibilityRole="button"
          accessibilityLabel="Open recent message"
          onPress={() => router.push(chatHref(recent.id))}
          className="mt-4 flex-row items-center gap-3 rounded-2xl bg-surface p-4 shadow-sm shadow-black/5"
        >
          <Avatar className="size-11">
            <AvatarFallback className="bg-primary-soft">
              <Text className="font-sans text-sm font-semibold text-primary">
                {initials(other?.displayName ?? recent.subject ?? "S")}
              </Text>
            </AvatarFallback>
          </Avatar>
          <View className="min-w-0 flex-1">
            <Text
              className="font-sans text-base font-semibold text-ink"
              numberOfLines={1}
            >
              {other?.displayName ?? recent.subject ?? "Support"}
            </Text>
            <Text
              className="mt-0.5 font-sans text-sm text-muted"
              numberOfLines={1}
            >
              {recent.lastMessagePreview} ·{" "}
              {formatRelativeTime(recent.updatedAt)}
            </Text>
          </View>
          <ChevronRight color={colors.muted} size={18} />
        </Pressable>
      ) : null}

      <Pressable
        accessibilityRole="button"
        accessibilityLabel="Search for help"
        onPress={() => router.push("/(contact)/help")}
        className="mt-4 flex-row items-center gap-3 rounded-2xl bg-surface px-4 py-3.5 shadow-sm shadow-black/5"
      >
        <Search color={colors.muted} size={20} />
        <Text className="font-sans text-base text-muted">Search for help</Text>
      </Pressable>

      <OffersCarousel />

      <Pressable
        accessibilityRole="button"
        accessibilityLabel="Frequently asked questions"
        onPress={() => router.push("/(contact)/help")}
        className="mt-6 flex-row items-center justify-between rounded-2xl bg-surface p-4 shadow-sm shadow-black/5"
      >
        <View>
          <Text className="font-sans text-base font-semibold text-ink">
            FAQ
          </Text>
          <Text className="mt-1 font-sans text-sm text-muted">
            Common questions
          </Text>
        </View>
        <ChevronRight color={colors.muted} size={18} />
      </Pressable>
    </ScrollView>
  );
}
```

- [ ] **Step 1: Create `components/hub/faq-list.tsx`**

```tsx
import { ChevronDown, ChevronUp } from "lucide-react-native";
import { useState } from "react";
import { FlatList, Pressable, View } from "react-native";

import { Text } from "@/components/ui/text";
import { FAQ_ITEMS } from "@/lib/mock-data";
import { colors } from "@/lib/tokens";

export function FaqList() {
  const [openId, setOpenId] = useState<string | null>(FAQ_ITEMS[0]?.id ?? null);

  return (
    <FlatList
      className="flex-1 bg-bg"
      contentContainerClassName="px-4 py-4"
      data={FAQ_ITEMS}
      keyExtractor={(item) => item.id}
      ItemSeparatorComponent={() => <View className="h-3" />}
      renderItem={({ item }) => {
        const open = item.id === openId;
        return (
          <Pressable
            accessibilityRole="button"
            accessibilityLabel={item.question}
            accessibilityState={{ expanded: open }}
            onPress={() => setOpenId(open ? null : item.id)}
            className="rounded-2xl bg-surface p-4 shadow-sm shadow-black/5"
          >
            <View className="flex-row items-start justify-between gap-3">
              <Text className="flex-1 font-sans text-base font-semibold text-ink">
                {item.question}
              </Text>
              {open ? (
                <ChevronUp color={colors.muted} size={20} />
              ) : (
                <ChevronDown color={colors.muted} size={20} />
              )}
            </View>
            {open ? (
              <Text className="mt-2 font-sans text-sm leading-5 text-ink-soft">
                {item.answer}
              </Text>
            ) : null}
          </Pressable>
        );
      }}
    />
  );
}
```
